$port = New-Object System.IO.Ports.SerialPort COM3,115200,None,8,one
$port.Open()

$cmd  = [byte]0x00 # Cifrar 0 - Descifrar 1
$key  = [byte[]](0x00,0x00,0x00,0x00,0x00,0x00,0x00,0x00,0x00,0x00,0x00,0x00,0x00,0x00,0x00,0x00)
$data = [byte[]](0x00,0x00,0x00,0x00,0x00,0x00,0x00,0x00,0x00,0x00,0x00,0x00,0x00,0x00,0x00,0x00)

$trama = [byte[]]((,$cmd) + $key + $data)
$port.Write($trama, 0, $trama.Length)

Start-Sleep -Milliseconds 200
$resultado = New-Object byte[] 16
$port.Read($resultado, 0, 16) | Out-Null
$port.Close()

($resultado | ForEach-Object { "{0:x2}" -f $_ }) -join ""
